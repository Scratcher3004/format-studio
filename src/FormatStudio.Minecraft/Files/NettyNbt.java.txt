package {{namespace}};

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import java.io.IOException;
import net.minecraft.nbt.Tag;
import net.minecraft.network.codec.ByteBufCodecs;
import {{runtime}}.FormatReader;
import {{runtime}}.FormatWriter;

/**
 * NBT as Minecraft's tags (network NBT: an unnamed root), read and written by the generated read and write code: the
 * tag's bytes are found with NbtCodec and decoded with ByteBufCodecs.TAG. The code of the nbt native type for the netty
 * generator; StreamCodec.composite uses ByteBufCodecs.TAG directly.
 */
public final class NettyNbt {
    private NettyNbt() {
    }

    public static Tag readJavaNetwork(FormatReader reader) throws IOException {
        return ByteBufCodecs.TAG.decode(Unpooled.wrappedBuffer(NbtCodec.readJavaNetwork(reader)));
    }

    public static void writeJavaNetwork(FormatWriter writer, Tag tag) throws IOException {
        ByteBuf buf = Unpooled.buffer();
        ByteBufCodecs.TAG.encode(buf, tag);
        byte[] bytes = new byte[buf.readableBytes()];
        buf.readBytes(bytes);
        NbtCodec.writeJavaNetwork(writer, bytes);
    }
}
